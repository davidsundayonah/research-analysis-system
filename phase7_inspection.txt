import sqlite3
from pathlib import Path


class Database:
    def __init__(self):
        self.database_dir = Path("app/database")
        self.database_dir.mkdir(parents=True, exist_ok=True)

        self.database_path = self.database_dir / "research_analysis.db"

        self.connection = sqlite3.connect(self.database_path)
        self.connection.row_factory = sqlite3.Row

        self.create_tables()

    # =========================================================
    # DATABASE TABLES
    # =========================================================

    def create_tables(self):
        self.connection.execute("""
            CREATE TABLE IF NOT EXISTS projects (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                name TEXT NOT NULL,
                research_title TEXT,
                researcher TEXT,
                institution TEXT,
                department TEXT,
                significance_level REAL DEFAULT 0.05,
                status TEXT DEFAULT 'New',
                created_at TEXT DEFAULT CURRENT_TIMESTAMP,
                updated_at TEXT DEFAULT CURRENT_TIMESTAMP
            )
        """)

        self.connection.execute("""
            CREATE TABLE IF NOT EXISTS datasets (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                project_id INTEGER NOT NULL,
                filename TEXT NOT NULL,
                file_path TEXT NOT NULL,
                file_type TEXT NOT NULL,
                row_count INTEGER DEFAULT 0,
                column_count INTEGER DEFAULT 0,
                imported_at TEXT DEFAULT CURRENT_TIMESTAMP,

                FOREIGN KEY (project_id)
                    REFERENCES projects(id)
                    ON DELETE CASCADE
            )
        """)

        self.connection.commit()

    # =========================================================
    # PROJECTS
    # =========================================================

    def create_project(
        self,
        name,
        research_title="",
        researcher="",
        institution="",
        department="",
        significance_level=0.05,
    ):
        cursor = self.connection.execute(
            """
            INSERT INTO projects (
                name,
                research_title,
                researcher,
                institution,
                department,
                significance_level
            )
            VALUES (?, ?, ?, ?, ?, ?)
            """,
            (
                name,
                research_title,
                researcher,
                institution,
                department,
                significance_level,
            ),
        )

        self.connection.commit()

        return cursor.lastrowid

    def get_projects(self):
        cursor = self.connection.execute(
            """
            SELECT *
            FROM projects
            ORDER BY updated_at DESC
            """
        )

        return cursor.fetchall()

    def get_project(self, project_id):
        cursor = self.connection.execute(
            """
            SELECT *
            FROM projects
            WHERE id = ?
            """,
            (project_id,),
        )

        return cursor.fetchone()

    def delete_project(self, project_id):
        self.connection.execute(
            """
            DELETE FROM projects
            WHERE id = ?
            """,
            (project_id,),
        )

        self.connection.commit()

    # =========================================================
    # DATASETS
    # =========================================================

    def create_dataset(
        self,
        project_id,
        filename,
        file_path,
        file_type,
        row_count,

    ):
        cursor = self.connection.execute(

import pandas as pd
import numpy as np


class DataCleaner:

    def __init__(self, dataframe):
        self.dataframe = dataframe.copy()

    # =========================================================
    # RENAME VARIABLE
    # =========================================================

    def rename_variable(
        self,
        old_name,
        new_name
    ):

        if old_name not in self.dataframe.columns:
            raise ValueError(
                f"Variable '{old_name}' does not exist."
            )

        if not new_name:
            raise ValueError(
                "New variable name cannot be empty."
            )

        if new_name in self.dataframe.columns:
            raise ValueError(
                f"Variable '{new_name}' already exists."
            )

        self.dataframe = self.dataframe.rename(
            columns={
                old_name: new_name
            }
        )

        return self.dataframe

    # =========================================================
    # RECODE VALUES
    # =========================================================

    def recode_values(
        self,
        variable,
        mapping,
        new_variable=None
    ):

        if variable not in self.dataframe.columns:
            raise ValueError(
                f"Variable '{variable}' does not exist."
            )

        target = (
            new_variable
            if new_variable
            else variable
        )

        self.dataframe[target] = (
            self.dataframe[variable]
            .replace(mapping)
        )

        return self.dataframe

    # =========================================================
    # DEFINE MISSING VALUES
    # =========================================================

    def define_missing_values(
        self,
        variable,
        missing_values
    ):

        if variable not in self.dataframe.columns:
            raise ValueError(
                f"Variable '{variable}' does not exist."
            )

        self.dataframe[variable] = (
            self.dataframe[variable]
            .replace(
                missing_values,
                np.nan
            )
        )

        return self.dataframe

    # =========================================================
    # REVERSE SCORE
    # =========================================================

    def reverse_score(
        self,
        variable,
        minimum=1,
        maximum=5,
        new_variable=None
    ):

        if variable not in self.dataframe.columns:
            raise ValueError(
                f"Variable '{variable}' does not exist."
            )

        target = (
            new_variable
            if new_variable
            else variable
        )

        self.dataframe[target] = (
            maximum
            + minimum
            - self.dataframe[variable]
        )

        return self.dataframe

    # =========================================================
    # COMPUTED VARIABLE
    # =========================================================

    def compute_variable(
        self,
        new_variable,
        variables,
        operation="mean"
    ):

        missing = [
            variable
            for variable in variables
            if variable not in self.dataframe.columns
        ]

        if missing:
            raise ValueError(
                f"Variables not found: {missing}"
            )

        if operation == "mean":

            self.dataframe[new_variable] = (
                self.dataframe[variables]
                .mean(axis=1)
            )

        elif operation == "sum":

            self.dataframe[new_variable] = (
                self.dataframe[variables]
                .sum(axis=1)
            )

        elif operation == "min":

            self.dataframe[new_variable] = (
                self.dataframe[variables]
                .min(axis=1)
            )

        elif operation == "max":

            self.dataframe[new_variable] = (
                self.dataframe[variables]
                .max(axis=1)
            )

        else:

            raise ValueError(
                f"Unsupported operation: {operation}"
            )

        return self.dataframe

    # =========================================================
    # DUPLICATES
    # =========================================================

    def find_duplicates(self):

        return self.dataframe[
            self.dataframe.duplicated(
                keep=False
            )
        ]

    # =========================================================
    # REMOVE DUPLICATES
    # =========================================================

    def remove_duplicates(self):

        self.dataframe = (
            self.dataframe
            .drop_duplicates()
            .reset_index(drop=True)
        )

        return self.dataframe

    # =========================================================
    # MISSING VALUE SUMMARY
    # =========================================================

    def missing_summary(self):

        summary = []

        for column in self.dataframe.columns:

            missing = int(
                self.dataframe[column]
                .isna()
                .sum()
            )

            total = len(
                self.dataframe
            )

            percentage = (
                missing / total * 100
                if total > 0
                else 0
            )

            summary.append(
                {
                    "variable": column,
                    "missing": missing,
                    "percentage": percentage,
                }
            )

        return summary

    # =========================================================
    # NUMERIC RANGE CHECK
    # =========================================================

    def find_out_of_range(
        self,
        variable,
        minimum,
        maximum
    ):

        if variable not in self.dataframe.columns:
            raise ValueError(
                f"Variable '{variable}' does not exist."
            )

        numeric = pd.to_numeric(
            self.dataframe[variable],
            errors="coerce"
        )

        mask = (
            (numeric < minimum)
            | (numeric > maximum)
        )

        return self.dataframe[
            mask
        ]

    # =========================================================
    # GET DATAFRAME
    # =========================================================

    def get_dataframe(self):

        return self.dataframe.copy()

import pandas as pd
import numpy as np


class DescriptiveStatistics:

    def __init__(self, dataframe):
        """
        Initialize the descriptive statistics engine.

        Parameters
        ----------
        dataframe : pandas.DataFrame
            Dataset to analyze.
        """

        if not isinstance(dataframe, pd.DataFrame):
            raise TypeError(
                "dataframe must be a pandas DataFrame."
            )

        self.dataframe = dataframe.copy()

    # =========================================================
    # INTERNAL HELPERS
    # =========================================================

    def _validate_variable(self, variable):
        """
        Check that a variable exists in the dataset.
        """

        if variable not in self.dataframe.columns:
            raise ValueError(
                f"Variable '{variable}' does not exist."
            )

    def _numeric_series(self, variable):
        """
        Convert a variable to numeric values.

        Invalid values are converted to NaN.
        """

        self._validate_variable(variable)

        return pd.to_numeric(
            self.dataframe[variable],
            errors="coerce"
        )

    # =========================================================
    # NUMERIC SUMMARY
    # =========================================================

    def numeric_summary(self, variables=None):
        """
        Generate descriptive statistics for numeric variables.

        Returns
        -------
        pandas.DataFrame
        """

        if variables is None:
            variables = self.dataframe.columns.tolist()

        results = []

        for variable in variables:

            if variable not in self.dataframe.columns:
                continue

            series = pd.to_numeric(
                self.dataframe[variable],
                errors="coerce"
            )

            valid = series.dropna()

            if len(valid) == 0:
                continue

            mode_values = valid.mode()

            if len(mode_values) > 0:
                mode = mode_values.iloc[0]
            else:
                mode = np.nan

            results.append({
                "Variable": variable,

                "N": int(valid.count()),

                "Missing": int(
                    series.isna().sum()
                ),

                "Mean": round(
                    valid.mean(),
                    3
                ),

                "Median": round(
                    valid.median(),
                    3
                ),

                "Mode": round(
                    mode,
                    3
                ) if pd.notna(mode) else np.nan,

                "Std. Deviation": round(
                    valid.std(),
                    3
                ),

                "Variance": round(
                    valid.var(),
                    3
                ),

                "Minimum": round(
                    valid.min(),
                    3
                ),

                "Maximum": round(
                    valid.max(),
                    3
                ),

                "Range": round(
                    valid.max() - valid.min(),
                    3
                ),

                "Skewness": round(
                    valid.skew(),
                    3
                ) if len(valid) >= 3 else np.nan,

                "Kurtosis": round(
                    valid.kurt(),
                    3
                ) if len(valid) >= 4 else np.nan,

                "Q1": round(

                "Q2": round(

                ),

                "Q3": round(

    # =========================================================
    # FREQUENCY TABLE
    # =========================================================

    def frequency_table(self, variable):
        """
        Generate a frequency table.

        Columns:
        - Value
        - Frequency
        - Percent
        - Valid Percent
        - Cumulative Percent
        """

        self._validate_variable(variable)

        series = self.dataframe[variable]

        total = len(series)

        valid_series = series.dropna()

        valid_total = len(valid_series)

        frequencies = (
            series
            .value_counts(
                dropna=False
            )
        )

        results = []

        cumulative_valid = 0

        for value, frequency in frequencies.items():

            is_missing = pd.isna(value)

            if is_missing:

                label = "Missing"

                percent = (
                    frequency / total * 100
                    if total > 0
                    else 0
                )

                valid_percent = np.nan

                cumulative_percent = np.nan

            else:

                label = value

                percent = (
                    frequency / total * 100
                    if total > 0
                    else 0
                )

                valid_percent = (
                    frequency / valid_total * 100
                    if valid_total > 0
                    else 0
                )

                cumulative_valid += valid_percent

                cumulative_percent = cumulative_valid

            results.append({

                "Value": label,

                "Frequency": int(
                    frequency
                ),

                "Percent": round(
                    percent,
                    2
                ),

                "Valid Percent": (
                    round(
                        valid_percent,
                        2
                    )
                    if pd.notna(valid_percent)
                    else np.nan
                ),

                "Cumulative Percent": (
                    round(
                        cumulative_percent,
                        2
                    )
                    if pd.notna(
                        cumulative_percent
                    )
                    else np.nan
                ),
            })

        return pd.DataFrame(
            results,
            columns=[
                "Value",
                "Frequency",
                "Percent",
                "Valid Percent",
                "Cumulative Percent",
            ],
        )

    # =========================================================

    def frequency_tables(self, variables):
        """
        Generate frequency tables for multiple variables.

        Returns
        -------
        dict
            Dictionary where keys are variable names and
            values are frequency DataFrames.
        """

        tables = {}

        for variable in variables:

            tables[variable] = (
                self.frequency_table(
                    variable
                )
            )

        return tables

    def mode(self, variable):

    # PERCENTILES
    # =========================================================

    def percentiles(
        self,
        variable,
        percentiles=None
    ):
        """
        Calculate requested percentiles.

        Example
        -------
        percentiles(
            "Age",
            [10, 25, 50, 75, 90]
        )
        """

        if percentiles is None:
            percentiles = [
                10,
                25,
                50,
                75,
                90
            ]

        series = self._numeric_series(
            variable
        ).dropna()

        if len(series) == 0:
            return {}

        results = {}

        for percentile in percentiles:

            if not 0 <= percentile <= 100:
                raise ValueError(
                    "Percentiles must be between "
                    "0 and 100."
                )

            results[
                f"P{percentile}"
            ] = round(
                series.quantile(
                    percentile / 100
                ),
                3
            )

        return results

    # =========================================================
    # MISSING VALUE SUMMARY
    # =========================================================

    def missing_summary(self):
        """
        Generate a missing-value summary for
        every variable.
        """

        results = []

        total_rows = len(
            self.dataframe
        )

        for variable in self.dataframe.columns:

            missing = int(
                self.dataframe[
                    variable
                ].isna().sum()

            missing_percentage = (
                missing / total_rows * 100
                if total_rows > 0
                else 0
            )

            valid_percentage = (
                valid / total_rows * 100
                if total_rows > 0
                else 0
            )

            results.append({

                "Variable": variable,

                "Valid": valid,

                "Missing": missing,

                "Valid Percent": round(
                    valid_percentage,
                    2
                ),

                "Missing Percent": round(
                    missing_percentage,
                    2
                ),
            })

        return pd.DataFrame(results)

import pandas as pd
import numpy as np


class VariableInspector:

    def __init__(self, dataframe):
        self.dataframe = dataframe

    # =========================================================
    # INSPECT ALL VARIABLES
    # =========================================================

    def inspect(self):
        variables = []

        for column in self.dataframe.columns:

            series = self.dataframe[column]

            information = self.inspect_variable(
                column,
                series
            )

            variables.append(
                information
            )

        return variables

    # =========================================================
    # INSPECT ONE VARIABLE
    # =========================================================

    def inspect_variable(
        self,
        name,
        series
    ):

        variable_type = self.detect_type(
            series
        )

        measure = self.detect_measure(
            series
        )

        missing_count = int(
            series.isna().sum()
        )

        unique_count = int(
            series.nunique(
                dropna=True
            )
        )

        result = {
            "name": name,
            "label": name,
            "type": variable_type,
            "measure": measure,
            "missing": missing_count,
            "unique": unique_count,
            "count": int(
                series.count()
            ),
        }

        # -----------------------------------------------------
        # Numeric information
        # -----------------------------------------------------

        if pd.api.types.is_numeric_dtype(
            series
        ):

            result["minimum"] = self.safe_numeric(
                series.min()
            )

            result["maximum"] = self.safe_numeric(
                series.max()
            )

            result["mean"] = self.safe_numeric(
                series.mean()
            )

            result["std"] = self.safe_numeric(
                series.std()
            )

        else:

            result["minimum"] = None
            result["maximum"] = None
            result["mean"] = None
            result["std"] = None

        return result

    # =========================================================
    # DETECT TYPE
    # =========================================================

    def detect_type(
        self,
        series
    ):

        if pd.api.types.is_bool_dtype(
            series
        ):
            return "Boolean"

        if pd.api.types.is_numeric_dtype(
            series
        ):
            return "Numeric"

        if pd.api.types.is_datetime64_any_dtype(
            series
        ):
            return "Date"

        return "Text"

    # =========================================================
    # DETECT MEASUREMENT LEVEL
    # =========================================================

    def detect_measure(
        self,
        series
    ):

        # -----------------------------------------------------
        # Boolean
        # -----------------------------------------------------

        if pd.api.types.is_bool_dtype(
            series
        ):
            return "Nominal"

        # -----------------------------------------------------
        # Text
        # -----------------------------------------------------

        if not pd.api.types.is_numeric_dtype(
            series
        ):
            return "Nominal"

        clean = series.dropna()

        if len(clean) == 0:
            return "Unknown"

        unique_values = clean.nunique()

        # -----------------------------------------------------
        # Binary numeric variable
        # -----------------------------------------------------

        if unique_values == 2:
            return "Nominal"

        # -----------------------------------------------------
        # Likert-style variable
        # -----------------------------------------------------

        if (
            unique_values <= 7
            and clean.min() >= 1
            and clean.max() <= 7
        ):
            return "Ordinal"

        # -----------------------------------------------------
        # Small categorical numeric variable
        # -----------------------------------------------------

        if unique_values <= 10:
            return "Ordinal"

        # -----------------------------------------------------
        # Continuous variable
        # -----------------------------------------------------

        return "Scale"

    # =========================================================
    # SAFE NUMERIC VALUE
    # =========================================================

    def safe_numeric(
        self,
        value
    ):

        if pd.isna(value):
            return None

        if isinstance(
            value,
            (np.integer,)
        ):
            return int(value)

        if isinstance(
            value,
            (np.floating,)
        ):
            return float(value)

        return value

    def open_cleaning_dialog(self):

        if self.current_dataframe is None:

            QMessageBox.warning(
                self,
                "No Dataset",
                "Please import a dataset first.",
            )

            return

        dialog = DataCleaningDialog(
            self.current_dataframe,
            self
        )

        if dialog.exec() != QDialog.Accepted:
            return

        data = dialog.get_operation_data()

        cleaner = DataCleaner(
            self.current_dataframe
        )

        try:

            operation = data["operation"]

            variable = data["variable"]

            new_variable = (
                data["new_variable"]
                or None
            )

            if operation == "Rename Variable":

                if not new_variable:
                    raise ValueError(
                        "Enter a new variable name."
                    )

                dataframe = cleaner.rename_variable(
                    variable,
                    new_variable
                )

            elif operation == "Recode Values":

                mapping = self.parse_mapping(
                    data["mapping"]
                )

                dataframe = cleaner.recode_values(
                    variable,
                    mapping,
                    new_variable
                )

            elif operation == "Define Missing Values":

                mapping = self.parse_mapping(
                    data["mapping"]
                )

                missing_values = list(
                    mapping.keys()
                )

                dataframe = cleaner.define_missing_values(
                    variable,
                    missing_values
                )

            elif operation == "Reverse Score":

                dataframe = cleaner.reverse_score(
                    variable,
                    data["minimum"],
                    data["maximum"],
                    new_variable
                )

            else:

                raise ValueError(
                    "Unknown operation."
                )

            self.current_dataframe = dataframe

            self.display_dataframe(
                dataframe
            )

            self.display_variable_view(
                dataframe
            )

            QMessageBox.information(
                self,
                "Operation Complete",
                "The data was updated successfully.",
            )

        except Exception as error:

            QMessageBox.critical(
                self,
                "Cleaning Error",
                str(error),
            )

    def parse_mapping(self, text):

        if not text.strip():

            raise ValueError(
                "Enter a mapping."
            )

        mapping = {}

        parts = text.split(",")

        for part in parts:

            if "=" not in part:

                raise ValueError(
                    f"Invalid mapping: {part}"
                )

            old, new = part.split(
                "=",
                1
            )

            old = old.strip()
            new = new.strip()

            try:
                old_value = float(old)
            except ValueError:
                old_value = old

            try:
                new_value = float(new)
            except ValueError:
                new_value = new

            if isinstance(
                old_value,
                float
            ) and old_value.is_integer():

                old_value = int(
                    old_value
                )

            if isinstance(
                new_value,
                float
            ) and new_value.is_integer():

                new_value = int(
                    new_value
                )

            mapping[
                old_value
            ] = new_value

        return mapping

def reverse_score(
    self,
    variable,
    minimum=1,
    maximum=5,
    new_variable=None
):

    if variable not in self.dataframe.columns:
        raise ValueError(
            f"Variable '{variable}' does not exist."
        )

    target = (
        new_variable
        if new_variable
        else variable
    )

    self.dataframe[target] = (
        maximum
        + minimum
        - self.dataframe[variable]
    )

    return self.dataframe

import pandas as pd

from app.statistics.descriptive_statistics import (
    DescriptiveStatistics,

def test_engine_accepts_different_research_datasets():
    education_data = pd.DataFrame({
        "School": ["A", "B", "A", "C"],
        "Score": [60, 75, 80, 85],
    })

    business_data = pd.DataFrame({
        "Department": ["HR", "IT", "IT", "Sales"],
        "Revenue": [1200, 2500, 3000, 1800],
    })

    education_engine = DescriptiveStatistics(
        education_data
    )

    business_engine = DescriptiveStatistics(
        business_data
    )

    assert education_engine.variable_summary(
        "Score"
    )["Mean"] == 75

    assert business_engine.variable_summary(
        "Revenue"
    )["Mean"] == 2125

    assert "School" in (
        education_engine.frequency_tables(["School"])
    )

    assert "Department" in (
        business_engine.frequency_tables(["Department"])
    )


def test_engine_handles_different_column_names():
    data = pd.DataFrame({
        "Monthly_Expenses": [100, 200, 300],
        "Customer_Type": ["New", "Returning", "New"],
    })

    engine = DescriptiveStatistics(data)

    assert engine.variable_summary(
        "Monthly_Expenses"
    )["Mean"] == 200

    assert "New" in engine.frequency_table(
        "Customer_Type"
    )["Value"].tolist()


def test_engine_handles_missing_values():
    data = pd.DataFrame({
        "Score": [10, 20, None, 40],
        "Group": ["A", "B", None, "A"],
    })

    engine = DescriptiveStatistics(data)

    summary = engine.variable_summary("Score")

    assert summary["N"] == 3
    assert summary["Missing"] == 1

    frequencies = engine.frequency_table("Group")

    assert frequencies["Frequency"].sum() == 4


def test_variable_detection_respects_column_dtypes():
    data = pd.DataFrame({
        "Count": pd.Series([1, 2], dtype="int64"),
        "Rate": pd.Series([0.25, 0.75], dtype="float64"),
        "Optional_Count": pd.Series([1, None], dtype="Int64"),
        "Numeric_Text": ["1", "2"],
        "Group": ["control", "treatment"],
        "Enabled": [True, False],
        "Observed_At": pd.to_datetime([
            "2025-01-01",
            "2025-01-02",
        ]),
    })

    engine = DescriptiveStatistics(data)

    assert engine.numeric_variables() == [
        "Count",
        "Rate",
        "Optional_Count",
    ]
    assert engine.categorical_variables() == [
        "Numeric_Text",
        "Group",
        "Enabled",
        "Observed_At",
    ]


def test_numeric_text_can_be_summarized_without_being_detected_as_numeric():

    engine = DescriptiveStatistics(data)

    assert engine.numeric_variables() == []
    assert engine.categorical_variables() == ["Score_Text"]
